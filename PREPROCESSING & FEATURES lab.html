01

1. Text Cleaning & Normalization

Remove HTML tags (<p>), web links (https://...), punctuation (!, %), and lowercase all letters.

re.sub(pattern, '', text).lower()
02

2. NLTK Tokenization

Chop continuous text string into individual word tokens using NLTK word_tokenize.

nltk.word_tokenize(clean_text)
03

3. NLTK Stopword Removal

Filter out common filler words using NLTK english stopwords list.

[w for w in tokens if w not in stop]
04

4. NLTK Stemming

Fast chopping of word endings using NLTK PorterStemmer ('breached' -> 'breach', 'dropped' -> 'drop').

PorterStemmer().stem(w)
05

5. NLTK Lemmatization

Dictionary root word lookup using NLTK WordNetLemmatizer ('breached' -> 'breach', 'dropped' -> 'drop').

WordNetLemmatizer().lemmatize(w)
06

6. N-grams (NLTK ngrams)

Combine words into 2-word pairs using NLTK ngrams function ('service level', 'level agreement', 'sla breach').

list(ngrams(lemmas, 2))
07

7. Bag of Words (BoW) & TF-IDF

Transform cleaned tokens into numerical TF-IDF feature weights.

TfidfVectorizer().fit_transform(text)
08

8. Word Frequency & Keyword Extraction

Extract top keywords summarizing customer complaints using Counter.

Counter(lemmas).most_common(3)
09

9. Domain Vocabulary Expansion

Expand business acronyms ('SLA' -> 'Service Level Agreement', 'ROI' -> 'Return on Investment').

acronyms.get(word, word)
10

10. Business Feature Engineering

Extract numeric business risk signals (length, '!', urgency flag, churn risk).

features = {'length': len(review), ...}
11

11. Text Data Visualization

Generate a Word Cloud image visualizing customer review keywords.

WordCloud().generate(text)

# 3. Stopword Removal                9. Domain Vocabulary & Acronym Expansion

# Step 0: Install Dependencies & NLTK Downloads
!pip install -q nltk scikit-learn pandas numpy matplotlib wordcloud networkx

import re
from collections import Counter
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

import nltk
nltk.download('punkt', quiet=True)
nltk.download('punkt_tab', quiet=True)
nltk.download('stopwords', quiet=True)
nltk.download('wordnet', quiet=True)
nltk.download('averaged_perceptron_tagger', quiet=True)
nltk.download('averaged_perceptron_tagger_eng', quiet=True)

from nltk.tokenize import word_tokenize, sent_tokenize
from nltk.corpus import stopwords, wordnet
from nltk.stem import PorterStemmer, WordNetLemmatizer
from nltk.util import ngrams
from sklearn.feature_extraction.text import CountVectorizer, TfidfVectorizer
from wordcloud import WordCloud
import networkx as nx

# OUR REALISTIC MBA BENCHMARK CUSTOMER REVIEW PARAGRAPH
SINGLE_RAW_REVIEW = "<p>URGENT: Our Service Level Agreement (SLA) was breached due to late product delivery! Visit https://help.company.com for support. Customer Return on Investment (ROI) dropped by 20%. I demand an immediate refund or our company will churn NOW!!!</p>"

print("=== RAW INPUT MBA REVIEW ===")
print(SINGLE_RAW_REVIEW)

print("\n--- 1. TEXT CLEANING & NORMALIZATION ---")
def clean_and_normalize(text):
    text = re.sub(r'<[^>]+>', ' ', text)  # Strip HTML markup
    text = re.sub(r'https?://\S+|www\.\S+', ' ', text)  # Strip URLs
    text = re.sub(r'[^a-zA-Z0-9\s]', ' ', text)  # Replace punctuation with space
    return re.sub(r'\s+', ' ', text).lower().strip()  # Lowercase & normalize spaces

clean_review = clean_and_normalize(SINGLE_RAW_REVIEW)
print("Cleaned Review:\n", clean_review)

print("\n--- 2. TOKENIZATION ---")
word_tokens = word_tokenize(clean_review)
print("Word Tokens (Total", len(word_tokens), "):\n", word_tokens)

print("\n--- 3. STOPWORD REMOVAL ---")
english_stopwords = set(stopwords.words('english')) - {'not', 'no', 'never'}
custom_stopwords = english_stopwords.union({'visit', 'company', 'com', 'help', 'due'})
filtered_tokens = [w for w in word_tokens if w not in custom_stopwords]
print("Filtered Tokens (Total", len(filtered_tokens), "):\n", filtered_tokens)

print("\n--- 4. STEMMING (PORTER STEMMER) ---")
stemmer = PorterStemmer()
stemmed_tokens = [stemmer.stem(w) for w in filtered_tokens]
print("Stemmed Tokens:\n", stemmed_tokens)

print("\n--- 5. LEMMATIZATION (WORDNET POS-AWARE) ---")
lemmatizer = WordNetLemmatizer()
def get_wordnet_pos(word):
    tag = nltk.pos_tag([word])[0][1][0].upper()
    return {"J": wordnet.ADJ, "N": wordnet.NOUN, "V": wordnet.VERB, "R": wordnet.ADV}.get(tag, wordnet.NOUN)

lemmatized_tokens = [lemmatizer.lemmatize(w, get_wordnet_pos(w)) for w in filtered_tokens]
print("Lemmatized Tokens:\n", lemmatized_tokens)

print("\n--- 6. N-GRAMS & PHRASE DETECTION ---")
bigrams = list(ngrams(lemmatized_tokens, 2))
print("Bigrams:\n", bigrams[:5])

print("\n--- 7. BAG OF WORDS (BoW) & TF-IDF VECTORIZATION ---")
processed_doc = " ".join(lemmatized_tokens)
corpus = [processed_doc, "delivery was fast product quality is good excellent customer service"]

bow_vec = CountVectorizer()
bow_matrix = bow_vec.fit_transform([processed_doc])
bow_df = pd.DataFrame(bow_matrix.toarray(), columns=bow_vec.get_feature_names_out())
print("BoW Word Frequencies:\n", bow_df.T[bow_df.T[0] > 0])

tfidf_vec = TfidfVectorizer()
tfidf_matrix = tfidf_vec.fit_transform(corpus)
tfidf_df = pd.DataFrame(tfidf_matrix.toarray(), columns=tfidf_vec.get_feature_names_out())
print("\nTop TF-IDF Scores for MBA Review:\n", tfidf_df.iloc[0].sort_values(ascending=False).head(5))

print("\n--- 8. WORD FREQUENCY & KEYWORD EXTRACTION ---")
freq_dist = Counter(lemmatized_tokens)
print("Top Word Frequencies:", freq_dist.most_common(5))

print("\n--- 9. DOMAIN-SPECIFIC BUSINESS VOCABULARY ---")
BUSINESS_ACRONYMS = {"sla": "service level agreement", "roi": "return on investment"}
expanded_words = [BUSINESS_ACRONYMS.get(w, w) for w in clean_review.split()]
print("Acronym Expanded Review:\n", " ".join(expanded_words))

print("\n--- 10. FEATURE ENGINEERING FOR BUSINESS TEXT ---")
features = {
    'char_length': len(SINGLE_RAW_REVIEW),
    'word_count': len(word_tokens),
    'uppercase_ratio': round(sum(1 for c in SINGLE_RAW_REVIEW if c.isupper()) / len(SINGLE_RAW_REVIEW), 3),
    'exclamation_count': SINGLE_RAW_REVIEW.count('!'),
    'has_urgency_flag': 1 if any(w in SINGLE_RAW_REVIEW for w in ['URGENT', 'NOW', 'SLA']) else 0,
    'churn_risk_flag': 1 if 'churn' in clean_review or 'refund' in clean_review else 0
}
print("Tabular Feature Vector:\n", pd.Series(features))

print("\n--- 11. TEXT DATA VISUALIZATION ---")
# 11a. WordCloud
wc = WordCloud(width=800, height=400, background_color='black', colormap='Greens').generate(processed_doc)
plt.figure(figsize=(10, 4))
plt.imshow(wc, interpolation='bilinear')
plt.axis('off')
plt.title('MBA Customer Review WordCloud')
plt.show()

# 11b. Term Network
G = nx.Graph()
for i in range(len(lemmatized_tokens) - 1):
    G.add_edge(lemmatized_tokens[i], lemmatized_tokens[i+1])

plt.figure(figsize=(8, 5))
pos = nx.spring_layout(G, k=0.5)
nx.draw_networkx_nodes(G, pos, node_color='#216b57', node_size=1200)
nx.draw_networkx_labels(G, pos, font_color='white', font_weight='bold')
nx.draw_networkx_edges(G, pos, edge_color='#e98249', width=2)
plt.title('MBA Review Term Co-occurrence Network')
plt.axis('off')
plt.show()
print("Pipeline execution completed for MBA review!")

from nltk.stem import PorterStemmer, WordNetLemmatizer
from nltk.util import ngrams
from sklearn.feature_extraction.text import CountVectorizer, TfidfVectorizer
from wordcloud import WordCloud
import networkx as nx

# OUR SINGLE BENCHMARK CUSTOMER REVIEW PARAGRAPH

print("=== RAW INPUT REVIEW ===")
print(SINGLE_RAW_REVIEW)

print("\n--- 1. TEXT CLEANING & NORMALIZATION ---")
def clean_and_normalize(text):
    text = re.sub(r'<[^>]+>', ' ', text)  # Strip HTML markup
    text = re.sub(r'https?://\S+|www\.\S+', ' ', text)  # Strip URLs
    text = re.sub(r'[\w.+-]+@[\w-]+\.[\w.-]+', ' ', text)  # Strip Emails
    text = re.sub(r'[^a-zA-Z0-9\s]', ' ', text)  # Replace punctuation with space
    return re.sub(r'\s+', ' ', text).lower().strip()  # Lowercase & normalize spaces

print("Sentence Tokens:\n", sentence_tokens)

print("Stemmed Tokens:\n", stemmed_tokens)

tfidf_matrix = tfidf_vec.fit_transform(corpus)

print("\nTop TF-IDF Scores for our Single Review:\n", tfidf_df.iloc[0].sort_values(ascending=False).head(5))
